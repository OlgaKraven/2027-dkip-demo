async (page) => {
 const results=[];
 for(const [stack,port] of [['mysql',5271],['postgresql',5272]]){
  for(const [offset,path,status,name] of [
   [0,'/notes',200,'api-response'],
   [0,'/notes?user_id=2147483647',200,'api-empty'],
   [0,'/notes?user_id=abc',400,'api-400'],
   [10,'/notes',500,'api-500'],
   [0,'/openapi.json',200,'openapi']
  ]){
   const response=await page.goto('http://127.0.0.1:'+(port+offset)+path);
   if(response.status()!==status)throw new Error(name+' '+response.status());
   await page.screenshot({path:'materials/screenshots/'+stack+'/'+name+'.png'});
   results.push({stack,path,status,body:await response.json()});
  }
 }
 return results;
}
